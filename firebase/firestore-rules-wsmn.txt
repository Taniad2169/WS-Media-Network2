    // ---------- START WSMN ----------
    function isWsmnOwner() {
      return request.auth != null
        && request.auth.token.email == 'thewebswann@gmail.com'
        && request.auth.token.email_verified == true;
    }

    // Provider profiles (doctors & attorneys, incl. up to 3 resized photos)
    // Public can read only ACTIVE listings. Only the owner can add/edit/pause/delete.
    match /providers/{id} {
      allow read: if resource.data.active == true || isWsmnOwner();
      allow create, update, delete: if isWsmnOwner();
    }

    // Visitor reviews left on the WSMN directory
    // Public can read only APPROVED reviews and can submit a new review,
    // which always arrives unapproved. Only the owner approves or deletes.
    match /reviews/{id} {
      allow read: if resource.data.approved == true || isWsmnOwner();
      allow create: if request.resource.data.keys().hasOnly(
                         ['providerId', 'providerName', 'name', 'stars', 'text', 'approved', 'createdAt'])
                    && request.resource.data.approved == false
                    && request.resource.data.providerId is string
                    && request.resource.data.providerId.size() > 0
                    && request.resource.data.providerId.size() < 100
                    && request.resource.data.name is string
                    && request.resource.data.name.size() <= 40
                    && request.resource.data.text is string
                    && request.resource.data.text.size() <= 600
                    && request.resource.data.stars is int
                    && request.resource.data.stars >= 1
                    && request.resource.data.stars <= 5
                    && request.resource.data.createdAt == request.time;
      allow update, delete: if isWsmnOwner();
    }
    // ---------- END WSMN ----------
